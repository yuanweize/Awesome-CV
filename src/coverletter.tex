%!TEX TS-program = lualatex
%!TEX encoding = UTF-8 Unicode
% Awesome CV LaTeX Template - Cover Letter
%
% Privacy-First Architecture:
% - workspace/current/config.tex: Personal info (shared with Resume)
% - workspace/current/sections/letter_body.tex: Letter content (gitignored)
%

%-------------------------------------------------------------------------------
% CONFIGURATIONS
%-------------------------------------------------------------------------------
\documentclass[11pt, a4paper]{awesome-cv}
\usepackage{modern-resume}

\geometry{left=1.65cm, top=1.25cm, right=1.65cm, bottom=1.45cm, footskip=.5cm}

% Color for highlights
\colorlet{awesome}{modernNavy}

% Section color highlighting
\setbool{acvSectionColorHighlight}{false}

% Social information separator
\renewcommand{\acvHeaderSocialSep}{\quad\textbar\quad}


%-------------------------------------------------------------------------------
% PERSONAL INFORMATION (shared with Resume)
%-------------------------------------------------------------------------------
\input{workspace/current/config.tex}


%-------------------------------------------------------------------------------
% LETTER INFORMATION
% Load from workspace/current/letter_config.tex (gitignored - per-application data)
%-------------------------------------------------------------------------------
\input{workspace/current/letter_config.tex}


%-------------------------------------------------------------------------------
\begin{document}

% Print the header (same style as Resume)
\makecvheader[L]

% Print the footer
\makeatletter
\makecvfooter
  {}
  {}
  {}
\makeatother

% Print the title
\makelettertitle

%-------------------------------------------------------------------------------
% LETTER CONTENT
%-------------------------------------------------------------------------------
\input{workspace/current/sections/letter_body.tex}

%-------------------------------------------------------------------------------
% Print the signature and enclosures
\makeletterclosing

\end{document}
